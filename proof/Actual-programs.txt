Define guess(tuvwx)=Func
Local n,guesses,k,h,beads,z
guesses:={}
For n,1,n_beads
  guesses:=augment(guesses,{uv(tuvwx,n)})
EndFor
If guesses≠tuvwx Then
  Disp "guess fn→input err":Return 
EndIf
z:=guesses
beads:=code
For k,1,n_beads
 If guesses[k]=code[k] Then
   guesses[k]:=colrightloc
   beads[k]:="."
EndIf
EndFor
For k,1,n_beads
 For h,1,n_beads
  If guesses[h]=beads[k] Then
    guesses[h]:=colwrongloc
    beads[k]:="?"
  EndIf
 EndFor
EndFor
For h,1,m
 For k,1,n_beads
  If a[h]=beads[k] Then
    beads[k]:="_"
  EndIf
  If a[h]=guesses[k] Then
    guesses[k]:=""
  EndIf
 EndFor
EndFor
Return {guesses,z}
EndFunc

Define pegin(status)=Prgm
    Local cpwp
    For cpwp,1,dim(status)[2]
            If status[1,cpwp]=colwrongloc Then
              Local indx
               For indx,1,dim(status)[2]
                  If where_loc[indx]≠indx Then
                    status[2,indx]:=status[2,cpwp]
                  EndIf
               EndFor
               status:=guess(mat▶list(status[2]))
            EndIf
    EndFor
    Local incr
    For incr,1,dim(status)[2]
      If status[1,incr]=colrightloc Then
        where_loc[incr]:=incr
        to_guess[incr]:=status[2,incr]
        If where_loc=seq(s,s,1,n_beads) Then
          Disp "found code",to_guess
          Stop
        Else
          Disp "Locking to→",where_loc
        EndIf
      EndIf
  EndFor
  status:=guess(rn_c())
  pegin(status)
EndPrgm

Define remote_qa(status,goal,rationale)=Prgm
Local cdx,qa,remote,reply,found
found:=goal
remote:=newcolours()
Request "Guess",remote,1
For cdx,1,n_beads
   For qa,1,3
      reply:="Y/N"
      Request rationale[qa,2]&string(cdx)&remote[cdx],reply,1
      If reply="Y" Then
        If rationale[qa,2]="✓" Then
           status[1,cdx]:=rationale[qa,2]
           status[2,cdx]:=remote[cdx]
        EndIf
        If status[1,cdx]≠"✓" Then
             status[1,cdx]:=rationale[qa,2]
             status[2,cdx]:=remote[cdx]
        Else
        EndIf
        Disp "status",status
        If status[1]=found Then
           Stop
        EndIf
        Exit
      EndIf
   EndFor
EndFor
  If status[1]=goal Then
    Stop
  EndIf
remote_qa(status,found,rationale)
EndPrgm

Define remote_fix(settings,status,remote,mmem)=Prgm
  Local r_n
  r_n:={"RANDOM","HOLD","TRIAL","SEQUENCE","ELIMINATION"}
  If settings=r_n[1] Then
    Define random=remote
    remote_run:=r_n[1]
  EndIf
  If settings=r_n[2] Then
    Define hold=rf_hold(remote,status)
    remote_run:=r_n[2]
  EndIf
  If settings=r_n[3] Then
    Define trial=rf_trial(remote,status)
    remote_run:=r_n[3]
  EndIf
  If settings=r_n[4] Then
    remote_fix(r_n[1],status,remote,mmem):remote:=expr(remote_run)
    remote_fix(r_n[5],status,remote,mmem):remote:=expr(remote_run)
    remote_fix(r_n[2],status,remote,mmem):remote:=expr(remote_run)
    remote_fix(r_n[3],status,remote,mmem):remote:=expr(remote_run)
    Define sequence=remote
    remote_run:=r_n[4]
  EndIf
  If settings=r_n[5] Then
    Define eliminate=rf_elimin8(remote,status,mmem)
    remote_run:=r_n[5]
  EndIf
  Define cmem=mmem
EndPrgm

Define rf_elimin8(remote,status,mmem)=Func
Local t,ccode,cmmem
  For t,1,dim(status[1])
    If status[1,t]="≠" Then
     cmmem:=newMat(1,mmem-1)
     For ccode,1,dim(mmem)
        If status[2,t]=mmem[1,ccode] Then
        Else
          cmmem[1,ccode]:=mmem[1,ccode]
        EndIf
     EndFor
     mmem:=cmmem
     remote[t]:=mmem[1,randInt(1,dim(mmem[1]),1)]
    Else
     remote[t]:=status[2,t]
    EndIf
  EndFor
  Return remote
EndFunc

Define rf_hold(remote,status)=Func
Local t
  For t,1,dim(status[1])[2]
    If status[1,t]="✓" Then
      remote[t]:=status[2,t]
    EndIf
  EndFor
  Return remote
EndFunc

Define rf_trial(remote,status)=Func
Local t,nf,ft
  For t,1,dim(status[1])
    If status[1,t]="%" Then
      ft:=randInt(1,dim(status[1]))
      nf:=status[ft]
      While "✓"=nf or ft=t
        ft:=randInt(1,dim(status[1]))
        nf:=status[ft]
        Exit
      EndWhile
      remote[t]:=status[2,t]
    EndIf
  EndFor
  Return remote
EndFunc
